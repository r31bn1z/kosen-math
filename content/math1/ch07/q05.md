$f(0) : (0)^2 - 2(0) = 0$

$f(2) : (2)^2 - 2(2) = 0$

$f(-3) : (-3)^2 - 2(-3) = 15$

$f(-x) : x^2 - 2x$

$f(x+1) : (x+1)^2 - 2(x+1) = x^2 + 2x + 1 - 2x - 2 = x^2 - 1$

$f(a+h) : (a+h)^2 - 2(a+h) = a^2 + 2ah + h^2 - 2a - 2h$

$f(a) : a^2 - 2a$

$f(a+h) - f(a) : 2ah + h^2 - 2h = h(2a + h - 2)$
