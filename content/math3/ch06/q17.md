### (1)
$$\begin{cases} x = \dfrac{1}{2}(u^2 - v^2) \\ y = uv \end{cases}$$

$$\begin{vmatrix} u & -v \\ v & u \end{vmatrix} = u^2 + v^2$$

### (2)
$$\begin{cases} x = u^2 + v^2 \\ y = uv \end{cases}$$

$$x_u = 2u \qquad x_v = 2v \qquad y_u = v \qquad y_v = u$$

$$\begin{vmatrix} 2u & 2v \\ v & u \end{vmatrix} = 2u^2 - 2v^2$$

### (3)
$$\begin{cases} x = \log\sqrt{u^2 + v^2} \\ y = \tan^{-1}\dfrac{v}{u} \end{cases}$$

$$x_u = \dfrac{u}{u^2+v^2} \qquad x_v = \dfrac{v}{u^2+v^2} \qquad y_u = \dfrac{-v}{u^2+v^2} \qquad y_v = \dfrac{u}{u^2+v^2}$$

$$\begin{vmatrix} \dfrac{u}{u^2+v^2} & \dfrac{v}{u^2+v^2} \\ \dfrac{-v}{u^2+v^2} & \dfrac{u}{u^2+v^2} \end{vmatrix} = \dfrac{u^2+v^2}{(u^2+v^2)^2} = \dfrac{1}{u^2+v^2}$$

### (4)
$$\begin{cases} r = \sqrt{x^2+y^2} \\ \theta = \tan^{-1}\dfrac{y}{x} \end{cases}$$

$$r_x = \dfrac{x}{\sqrt{x^2+y^2}} \qquad r_y = \dfrac{y}{\sqrt{x^2+y^2}} \qquad \theta_x = \dfrac{-y}{x^2+y^2} \qquad \theta_y = \dfrac{x}{x^2+y^2}$$

$$\begin{vmatrix} \dfrac{x}{\sqrt{x^2+y^2}} & \dfrac{y}{\sqrt{x^2+y^2}} \\ \dfrac{-y}{x^2+y^2} & \dfrac{x}{x^2+y^2} \end{vmatrix} = \dfrac{x^2+y^2}{(x^2+y^2)^{\frac{3}{2}}} = \dfrac{1}{\sqrt{x^2+y^2}}$$

### (5)
$$
\begin{cases}
x = \cos u \\
y = \sin u \cdot \cos v \\
z = \sin u \cdot \sin v \cdot \cos w
\end{cases}
$$

$$x_u = -\sin u \quad x_v = 0 \quad x_w = 0 \quad y_u = \cos u \cdot \cos v \quad y_v = -\sin u \cdot \sin v \quad y_w = 0$$

$$z_u = \cos u \cdot \sin v \cdot \cos w \quad z_v = \sin u \cdot \cos v \cdot \cos w \quad z_w = -\sin u \cdot \sin v \cdot \sin w$$

$$
\begin{vmatrix}
-\sin u & 0 & 0 \\
\cos u \cdot \cos v & -\sin u \cdot \sin v & 0 \\
\cos u \cdot \sin v \cdot \cos w & \sin u \cdot \cos v \cdot \cos w & -\sin u \cdot \sin v \cdot \sin w
\end{vmatrix}
= -\sin^3 u \cdot \sin^2 v \cdot \sin w
$$

### (6)
$$
\begin{cases}
x = r \sin\theta \cdot \cos\phi \\
y = r \sin\theta \cdot \sin\phi \\
z = r \cos\theta
\end{cases}
$$

$$x_\theta = r\cos\theta \cdot \cos\phi \quad x_\phi = -r\sin\theta \cdot \sin\phi \quad x_r = \sin\theta \cdot \cos\phi$$

$$y_\theta = r\cos\theta \cdot \sin\phi \quad y_\phi = r\sin\theta \cdot \cos\phi \quad y_r = \sin\theta \cdot \sin\phi$$

$$z_\theta = -r\sin\theta \quad z_\phi = 0 \quad z_r = \cos\theta$$

$$
\begin{vmatrix}
r\cos\theta \cdot \cos\phi & -r\sin\theta \cdot \sin\phi & \sin\theta \cdot \cos\phi \\
r\cos\theta \cdot \sin\phi & r\sin\theta \cdot \cos\phi & \sin\theta \cdot \sin\phi \\
-r\sin\theta & 0 & \cos\theta
\end{vmatrix}
$$

$$= r^2\cos^2\theta \cdot \sin\theta \cdot \cos^2\phi + r^2\sin^3\theta \cdot \sin^2\phi + r^2\sin^3\theta \cdot \cos^2\phi + r^2\cos^2\theta \cdot \sin\theta \cdot \sin^2\phi$$

$$= r^2\sin\theta$$
