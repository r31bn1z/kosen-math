### (1)
$$
\begin{vmatrix}
x & 2 & -1 & 1 \\
y & 1 & 3 & -2 \\
z & 5 & 2 & 4 \\
1 & 1 & 1 & 1
\end{vmatrix} = 0
$$

$$
\begin{vmatrix}
x-1 & 1 & -2 & 0 \\
y+2 & 3 & 5 & 0 \\
z-4 & 1 & -2 & 0 \\
1 & 1 & 1 & 1
\end{vmatrix} = \begin{vmatrix}
x-1 & 1 & -2 \\
y+2 & 3 & 5 \\
z-4 & 1 & -2
\end{vmatrix} = -6(x-1) - 2(y+2) + 5(z-4) - 5(x-1) + 2(y+2) + 6(z-4)
$$

$$= -11x + 11z - 33 = 0 \quad x - z + 3 = 0$$

### (2)
$$
\begin{vmatrix}
x & 1 & -1 & 3 \\
y & 2 & 1 & 0 \\
z & 3 & -2 & 1 \\
1 & 1 & 1 & 1
\end{vmatrix} = 0
$$

$$
\begin{vmatrix}
x-3 & -2 & -4 & 0 \\
y & 2 & 1 & 0 \\
z-1 & 2 & -3 & 0 \\
1 & 1 & 1 & 1
\end{vmatrix} = \begin{vmatrix}
x-3 & -2 & -4 \\
y & 2 & 1 \\
z-1 & 2 & -3
\end{vmatrix} = -6(x-3) - 8y - 2(z-1) - 2(x-3) - 6y + 8(z-1)
$$

$$= -8x - 14y + 6z + 18 = 0 \quad 4x + 7y - 3z - 9 = 0$$

### (3)
$$
\begin{vmatrix}
x & 1 & 2 & 0 \\
y & 1 & 0 & -2 \\
z & 0 & -1 & 0 \\
1 & 1 & 1 & 1
\end{vmatrix} = 0
$$

$$
\begin{vmatrix}
x & 1 & 2 & 0 \\
y+2 & 3 & 2 & 0 \\
z & 0 & -1 & 0 \\
1 & 1 & 1 & 1
\end{vmatrix} = \begin{vmatrix}
x & 1 & 2 \\
y+2 & 3 & 2 \\
z & 0 & -1
\end{vmatrix} = -3x + 2z + (y+2) - 6z = 0
$$

$$3x - y + 4z - 2 = 0$$
