### (1)

$$
\begin{vmatrix} 6 & 3 & 2 & 4 \\ -6 & -1 & 2 & -2 \\ 3 & 5 & 6 & 3 \\ 8 & 4 & 0 & 1 \end{vmatrix}
=
\begin{vmatrix} -26 & -13 & 2 & 0 \\ 10 & 7 & 2 & 0 \\ -21 & -7 & 6 & 0 \\ 8 & 4 & 0 & 1 \end{vmatrix}
=
\begin{vmatrix} -26 & -13 & 2 \\ 10 & 7 & 2 \\ -21 & -7 & 6 \end{vmatrix}
=
\begin{vmatrix} -26 & -13 & 2 \\ 36 & 20 & 0 \\ 57 & 32 & 0 \end{vmatrix}
=2\begin{vmatrix} 36 & 20 \\ 57 & 32 \end{vmatrix}
$$

$$=2(36\cdot32-20\cdot57)=24$$

### (2)

$$
\begin{vmatrix} 3 & 2 & 5 & 1 \\ -3 & -1 & 1 & 2 \\ 2 & 1 & 1 & 2 \\ 4 & 3 & -2 & 1 \end{vmatrix}
=
\begin{vmatrix} 3 & 2 & 5 & 1 \\ -9 & -5 & -9 & 0 \\ -4 & -3 & -9 & 0 \\ 1 & 1 & -7 & 0 \end{vmatrix}
=-\begin{vmatrix} -9 & -5 & -9 \\ -4 & -3 & -9 \\ 1 & 1 & -7 \end{vmatrix}
=-\begin{vmatrix} -4 & 0 & -44 \\ -1 & 0 & -30 \\ 1 & 1 & -7 \end{vmatrix}
=\begin{vmatrix} -4 & -44 \\ -1 & -30 \end{vmatrix}
$$

$$=-4\cdot(-30)-(-1)\cdot(-44)=76$$

### (3)

$$
\begin{vmatrix} 1 & 1 & 0 & 3 \\ -1 & 3 & 4 & 2 \\ 2 & 3 & 5 & 4 \\ 3 & -2 & -2 & 2 \end{vmatrix}
=
\begin{vmatrix} 1 & 1 & 0 & 3 \\ 0 & 4 & 4 & 5 \\ 0 & 1 & 5 & -2 \\ 0 & -5 & -2 & -7 \end{vmatrix}
=
\begin{vmatrix} 4 & 4 & 5 \\ 1 & 5 & -2 \\ -5 & -2 & -7 \end{vmatrix}
=
\begin{vmatrix} 0 & -16 & 13 \\ 1 & 5 & -2 \\ 0 & 23 & -17 \end{vmatrix}
=-\begin{vmatrix} -16 & 13 \\ 23 & -17 \end{vmatrix}
$$

$$=-(-16\cdot(-17)-13\cdot23)=27$$

### (4)

$$
\begin{vmatrix} 0 & 2 & 2 & 1 \\ 2 & 0 & 2 & 1 \\ 2 & 2 & 0 & 1 \\ 1 & 1 & 1 & 0 \end{vmatrix}
=
\begin{vmatrix} 0 & 2 & 2 & 1 \\ 0 & -2 & 0 & 1 \\ 0 & 0 & -2 & 1 \\ 1 & 1 & 1 & 0 \end{vmatrix}
=-\begin{vmatrix} 2 & 2 & 1 \\ -2 & 0 & 1 \\ 0 & -2 & 1 \end{vmatrix}
=-(4+4+4)=-12
$$
